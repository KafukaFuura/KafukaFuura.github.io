let mathQueue=Promise.resolve();
function clearMath(target){
  if(!target || !window.MathJax || !MathJax.typesetClear) return;
  try{MathJax.typesetClear([target]);}catch(_e){}
}
function typesetMath(target){
  if(!target) return Promise.resolve();
  mathQueue=mathQueue.then(async()=>{
    let tries=0;
    while((!window.MathJax || !MathJax.typesetPromise) && tries<80){await new Promise(r=>setTimeout(r,50));tries++;}
    if(!window.MathJax || !MathJax.typesetPromise) return;
    if(MathJax.startup?.promise) await MathJax.startup.promise;
    await MathJax.typesetPromise([target]);
  }).catch(err=>console.warn('MathJax typeset error:',err));
  return mathQueue;
}
